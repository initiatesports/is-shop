function coachAct(id, action) {
  if (!coachOk_()) return { ok: false, msg: '沒有權限' };
  var lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    var sh = shopSheet_(), last = sh.getLastRow();
    var ids = sh.getRange(2, 2, Math.max(last - 1, 1), 1).getValues(), r = -1;
    for (var i = 0; i < ids.length; i++) if (ids[i][0] === id) { r = i + 2; break; }
    if (r < 0) return { ok: false, msg: '找不到訂單 ' + id };
    var v = sh.getRange(r, 1, 1, 17).getValues()[0], email = String(v[15] || '').trim(), st = sh.getRange(r, 14);
    if (action === 'confirm' || action === 'resend') {
      if (action === 'confirm') st.setValue('已確認');
      if (!email) return { ok: true, msg: '訂單已確認（此訂單沒有電郵，未有發送通知）' };
      MailApp.sendEmail({ to: email, name: 'INITIATE SPORTS', subject: 'INITIATE SPORTS｜訂單確認 ' + id, htmlBody: confirmHtml_(v) });
      sh.getRange(r, 17).setValue(new Date());
      return { ok: true, msg: '已確認並發送電郵至 ' + email };
    }
    if (action === 'deliver') { st.setValue('已交收'); return { ok: true, msg: '已標記為已交收' }; }
    if (action === 'undeliver') { st.setValue('已確認'); return { ok: true, msg: '已撤回為已確認' }; }
    if (action === 'cancel') { st.setValue('已取消'); return { ok: true, msg: '訂單已取消' }; }
    if (action === 'reopen') { st.setValue(String(v[10]).indexOf('FPS') === 0 ? '待核實付款' : '待確認（現金）'); return { ok: true, msg: '訂單已恢復' }; }
    return { ok: false, msg: '未知操作' };
  } finally { lock.releaseLock(); }
}
